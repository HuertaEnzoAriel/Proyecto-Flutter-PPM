// Agregar un nuevo recordatorio, convirtiendo el string a DateTime
Future<void> agregarRecordatorio(
    List recordatorios, String titulo, String fechaTexto) async {
  DateTime fecha = DateTime.parse(fechaTexto); // "yyyy-MM-dd HH:mm" -> DateTime
  recordatorios.add({
    'titulo': titulo,
    'fecha': fecha.toString(),
  });
  await Future.delayed(Duration(seconds: 1));
  print("Recordatorio '$titulo' añadido.");
}

// Eliminar un recordatorio dado su título
Future<void> eliminarRecordatorio(List recordatorios, String titulo) async {
  recordatorios.removeWhere((r) => r['titulo'] == titulo);
  await Future.delayed(Duration(seconds: 1));
  print("Recordatorio '$titulo' eliminado.");
}
