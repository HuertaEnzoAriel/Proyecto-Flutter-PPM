import 'dart:convert';

Future<Map<String, dynamic>> leerJson() async {
  const contenidoArchivo = '''
  {
    "nombreApp": "Componentes",
    "rutas": [
      { "ruta": "alert", "icon": "add_alert", "texto": "Alertas" },
      { "ruta": "avatar", "icon": "accessibility", "texto": "Avatars" },
      { "ruta": "card", "icon": "folder_open", "texto": "Cards - Tarjetas" },
      { "ruta": "animatedContainer", "icon": "donut_large", "texto": "Animated Container" },
      { "ruta": "inputs", "icon": "input", "texto": "Inputs" },
      { "ruta": "slider", "icon": "tune", "texto": "Slider - Checks" },
      { "ruta": "list", "icon": "list", "texto": "Listas y Scroll" }
    ]
  }
  ''';

  await Future.delayed(Duration(milliseconds: 300)); // simula I/O
  return jsonDecode(contenidoArchivo);
}


Future<void> escribirJson(Map<String, dynamic> datos) async {
  await Future.delayed(Duration(milliseconds: 300));
  print('(Simulacion) Archivo actualizado correctamente.');
}


void mostrarRutas(List rutas) {
  print('--- Rutas ---');
  for (var r in rutas) {
    print('${r['ruta']} | icono: ${r['icon']} | texto: ${r['texto']}');
  }
}


Future<void> agregarRuta(
    List rutas, String ruta, String icon, String texto) async {
  rutas.add({
    'ruta': ruta,
    'icon': icon,
    'texto': texto,
  });
  await Future.delayed(Duration(seconds: 1));
  print("Ruta '$ruta' añadida.");
}


Future<void> modificarRuta(
    List rutas, String ruta, String nuevoIcon, String nuevoTexto) async {
  final index = rutas.indexWhere((r) => r['ruta'] == ruta);
  if (index == -1) {
    print("Ruta '$ruta' no encontrada.");
    return;
  }
  rutas[index]['icon'] = nuevoIcon;
  rutas[index]['texto'] = nuevoTexto;
  await Future.delayed(Duration(seconds: 1));
  print("Ruta '$ruta' modificada.");
}


Future<void> eliminarRuta(List rutas, String ruta) async {
  rutas.removeWhere((r) => r['ruta'] == ruta);
  await Future.delayed(Duration(seconds: 1));
  print("Ruta '$ruta' eliminada.");
}

void main() async {
  Map<String, dynamic> datos = await leerJson();
  List rutas = datos['rutas'];

  mostrarRutas(rutas);

  await agregarRuta(rutas, 'settings', 'settings', 'Configuracion');

  await modificarRuta(rutas, 'avatar', 'face', 'Perfil de usuario');

  await eliminarRuta(rutas, 'slider');

  await escribirJson(datos);

  print('--- JSON actualizado ---');
  print(JsonEncoder.withIndent('  ').convert(datos));
}
